cmake_minimum_required(VERSION 3.16)
project(ml_scan_context)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_package(PCL REQUIRED COMPONENTS common filters io)
find_package(OpenCV REQUIRED)
find_package(pcl_conversions REQUIRED)
add_library(scan_context SHARED src/Scancontext.cpp)
target_compile_features(scan_context PUBLIC cxx_std_17)
target_include_directories(scan_context PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(scan_context pcl_conversions)
target_link_libraries(scan_context Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
install(TARGETS scan_context EXPORT export_scan_context LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_scan_context HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 PCL OpenCV pcl_conversions)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
# 新核心使用Eigen和OpenMP；不引入ROS/PCL运行时逻辑。
add_library(polar_context SHARED src/polar_context.cpp src/database.cpp src/database_io.cpp src/preprocessing.cpp)
target_compile_features(polar_context PUBLIC cxx_std_17)
target_include_directories(polar_context PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(polar_context PUBLIC Eigen3::Eigen PRIVATE OpenMP::OpenMP_CXX)
install(TARGETS polar_context EXPORT export_polar_context LIBRARY DESTINATION lib)
ament_export_targets(export_polar_context HAS_LIBRARY_TARGET)
# PCD读取仅属于离线工具，不进入polar_context公共依赖。
add_executable(build_polar_database tools/build_polar_database.cpp)
target_include_directories(build_polar_database PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(build_polar_database PRIVATE polar_context ${PCL_LIBRARIES})
install(TARGETS build_polar_database DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME builder_test COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/builder_test.py $<TARGET_FILE:build_polar_database>)
  add_executable(polar_database_test test/polar_database_test.cpp)
  target_link_libraries(polar_database_test polar_context)
  add_test(NAME polar_database_test COMMAND polar_database_test)
  add_executable(preprocessing_test test/preprocessing_test.cpp)
  target_link_libraries(preprocessing_test polar_context)
  add_test(NAME preprocessing_test COMMAND preprocessing_test)
endif()
install(FILES UPSTREAM.md DESTINATION share/${PROJECT_NAME})
ament_package()
