2. 构建与自动测试（不启动车辆）
----------------------------
colcon build --symlink-install --executor sequential \
  --packages-select chassis_driver fast_lio ml_scan_context ml_relocalization \
    ml_icp_localization ml_localization_manager sc_pgo ml_bringup my_nav2_config \
  --cmake-args -DCMAKE_BUILD_TYPE=Release -DBUILD_TESTING=ON

colcon test --executor sequential \
  --packages-select ml_relocalization ml_icp_localization ml_localization_manager \
  --event-handlers console_direct+ --return-code-on-test-failure
colcon test-result --verbose
