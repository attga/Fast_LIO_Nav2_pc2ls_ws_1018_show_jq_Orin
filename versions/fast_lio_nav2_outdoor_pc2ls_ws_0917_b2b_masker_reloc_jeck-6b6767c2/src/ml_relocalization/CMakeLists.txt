cmake_minimum_required(VERSION 3.16)
project(ml_relocalization)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(ml_scan_context REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration)
find_package(Threads REQUIRED)
add_library(relocalization_core STATIC src/keyframe_map.cpp src/relocalizer.cpp src/registration.cpp)
target_compile_features(relocalization_core PUBLIC cxx_std_17)
target_include_directories(relocalization_core PUBLIC include ${PCL_INCLUDE_DIRS})
ament_target_dependencies(relocalization_core ml_scan_context)
target_link_libraries(relocalization_core ${PCL_LIBRARIES})
add_executable(relocalization_node src/relocalization_node.cpp)
ament_target_dependencies(relocalization_node rclcpp sensor_msgs geometry_msgs std_srvs pcl_conversions)
target_link_libraries(relocalization_node relocalization_core Threads::Threads)
install(TARGETS relocalization_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})

ament_package()
