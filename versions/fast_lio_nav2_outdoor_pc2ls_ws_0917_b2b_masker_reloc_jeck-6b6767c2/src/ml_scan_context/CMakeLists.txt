cmake_minimum_required(VERSION 3.16)
project(ml_scan_context)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters)
find_package(OpenCV REQUIRED)
find_package(pcl_conversions REQUIRED)
add_library(scan_context SHARED src/Scancontext.cpp)
target_compile_features(scan_context PUBLIC cxx_std_17)
target_include_directories(scan_context PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(scan_context pcl_conversions)
target_link_libraries(scan_context ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
install(TARGETS scan_context EXPORT export_scan_context LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_scan_context HAS_LIBRARY_TARGET)
ament_export_dependencies(PCL OpenCV pcl_conversions)
ament_package()
