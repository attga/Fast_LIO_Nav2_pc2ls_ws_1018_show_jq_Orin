cmake_minimum_required(VERSION 3.16)
project(ml_relocalization)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(ml_scan_context REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(Threads REQUIRED)
find_package(ml_ndt_localization REQUIRED)
add_library(relocalization_core STATIC src/keyframe_map.cpp src/relocalizer.cpp src/registration.cpp)
target_compile_features(relocalization_core PUBLIC cxx_std_17)
target_include_directories(relocalization_core PUBLIC include ${PCL_INCLUDE_DIRS})
ament_target_dependencies(relocalization_core ml_scan_context)
target_link_libraries(relocalization_core Eigen3::Eigen ${PCL_LIBRARIES} ml_ndt_localization::ml_autoware_ndt)
add_executable(relocalization_node src/relocalization_node.cpp)
ament_target_dependencies(relocalization_node rclcpp sensor_msgs geometry_msgs std_srvs pcl_conversions tf2_ros)
target_link_libraries(relocalization_node relocalization_core Threads::Threads)
install(TARGETS relocalization_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(registration_test test/registration_test.cpp)
  target_include_directories(registration_test PRIVATE src)
  target_link_libraries(registration_test relocalization_core)
  add_executable(replay_relocalization test/replay_relocalization.cpp)
  target_link_libraries(replay_relocalization relocalization_core)
  ament_add_gtest(matching_frame_test test/matching_frame_test.cpp)
  target_link_libraries(matching_frame_test relocalization_core)
  ament_add_gtest(confirmation_test test/confirmation_test.cpp)
  target_link_libraries(confirmation_test relocalization_core)
  ament_add_gtest(input_validation_test test/input_validation_test.cpp)
  target_link_libraries(input_validation_test relocalization_core)
endif()

ament_package()
